// --- PROCEDURAL NOISE & CAUSTICS ---
float hash(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
}

vec2 hash2(vec2 p) {
    return fract(sin(vec2(dot(p, vec2(127.1, 311.7)), dot(p, vec2(269.5, 183.3)))) * 43758.5453);
}

float noise(vec2 p) {
    vec2 i = floor(p);
    vec2 f = fract(p);
    vec2 u = f * f * (3.0 - 2.0 * f);
    return mix(
        mix(hash(i + vec2(0.0, 0.0)), hash(i + vec2(1.0, 0.0)), u.x),
        mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), u.x),
        u.y
    );
}

const mat2 m2 = mat2(0.8, -0.6, 0.6, 0.8);

// Animated cellular Voronoi pattern that mimics focused light caustics
float causticPattern(vec2 uv, float t) {
    vec2 i = floor(uv);
    vec2 f = fract(uv);
    float minDist = 1.0;
    
    for (int y = -1; y <= 1; y++) {
        for (int x = -1; x <= 1; x++) {
            vec2 neighbor = vec2(float(x), float(y));
            vec2 pt = hash2(i + neighbor);
            // Animate each cell point smoothly in a loop
            pt = 0.5 + 0.45 * sin(t * 1.4 + 6.2831 * pt);
            vec2 diff = neighbor + pt - f;
            minDist = min(minDist, length(diff));
        }
    }
    // Sharpen cell borders into bright light webs
    return pow(clamp(minDist, 0.0, 1.0), 4.0);
}

// Dual-layer caustics with RGB chromatic dispersion
vec3 getCaustics(vec2 p) {
    vec2 uv = p * 0.75;
    float t = u_time * 0.9;
    
    // Sample red, green, and blue at slightly offset coordinates for rainbow dispersion
    float r = causticPattern(uv + vec2(0.015, 0.0), t) + causticPattern(uv * 1.6 - t * 0.2, -t * 0.8) * 0.5;
    float g = causticPattern(uv, t)                      + causticPattern(uv * 1.6 - t * 0.2 + 0.01, -t * 0.8) * 0.5;
    float b = causticPattern(uv - vec2(0.015, 0.0), t) + causticPattern(uv * 1.6 - t * 0.2 - 0.01, -t * 0.8) * 0.5;
    
    return vec3(r, g, b) * 1.4;
}

// --- WATER & SEABED GEOMETRY ---
// Gentle, crystal-clear lagoon ripples
float waterHeight(vec2 p) {
    float h = 0.0;
    float amp = 0.5;
    float freq = 1.0;
    
    for (int i = 0; i < 5; i++) {
        vec2 q = p * freq;
        float w1 = sin(q.x * 1.4 + u_time * 1.6 + noise(q + u_time * 0.4) * 1.8);
        float w2 = cos(q.y * 1.3 - u_time * 1.3 + noise(q - u_time * 0.3) * 1.8);
        h += amp * (1.0 - abs(w1 * w2));
        
        p = m2 * p;
        freq *= 1.9;
        amp *= 0.48;
    }
    return h;
}

// Procedural sandy seabed with ripples and scattered pebbles
vec3 renderSeabed(vec3 hitPos, vec3 sunDir, float waterDepth) {
    // Wind/current sand ripples
    float sandRipple = sin(hitPos.x * 4.5 + noise(hitPos.xz * 1.5) * 3.0) * 0.5 + 0.5;
    float grain = noise(hitPos.xz * 28.0) * 0.08;
    
    // Warm golden sand color
    vec3 sandCol = mix(vec3(0.76, 0.67, 0.50), vec3(0.92, 0.84, 0.65), sandRipple) + grain;
    
    // Occasional darker coral rocks/pebbles on the bottom
    float rockMask = smoothstep(0.74, 0.77, noise(hitPos.xz * 0.9));
    vec3 rockCol = vec3(0.24, 0.28, 0.26) * (0.7 + 0.6 * noise(hitPos.xz * 8.0));
    sandCol = mix(sandCol, rockCol, rockMask);
    
    // Project caustics onto the seabed along the sun's refracted direction
    vec2 causticUV = hitPos.xz - sunDir.xz * (waterDepth / max(sunDir.y, 0.2));
    vec3 caustics = getCaustics(causticUV);
    
    // Combine base sand lighting with focused caustic light
    float diffuse = max(sunDir.y, 0.2);
    vec3 litFloor = sandCol * (diffuse * 0.7 + 0.3) + caustics * vec3(1.0, 0.96, 0.85) * 0.85;
    
    // Beer-Lambert Law: water absorbs red light faster than blue/green with depth
    vec3 absorption = exp(-vec3(0.42, 0.14, 0.08) * waterDepth * 1.4);
    vec3 deepScatter = vec3(0.02, 0.28, 0.36);
    
    return mix(deepScatter, litFloor, absorption);
}

// --- SKY ---
vec3 getSky(vec3 rd, vec3 sunDir) {
    float sunAmt = max(dot(rd, sunDir), 0.0);
    vec3 sky = mix(vec3(0.45, 0.78, 0.95), vec3(0.12, 0.45, 0.82), clamp(rd.y * 1.5, 0.0, 1.0));
    sky += vec3(1.0, 0.95, 0.85) * pow(sunAmt, 12.0) * 0.4;
    sky += vec3(1.0, 0.98, 0.90) * smoothstep(0.998, 0.9995, sunAmt) * 2.5;
    return sky;
}

// --- MAIN RENDER ---
void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    vec2 uv = (fragCoord * 2.0 - u_resolution.xy) / u_resolution.y;
    
    // Camera gliding over a shallow tropical reef
    vec3 ro = vec3(0.0, 2.8, u_time * 1.8);
    vec3 rd = normalize(vec3(uv.x, uv.y - 0.32, 1.4));
    
    // High afternoon tropical sun
    vec3 sunDir = normalize(vec3(0.35, 0.78, 0.52));
    
    vec3 col = vec3(0.0);
    
    if (rd.y >= 0.0) {
        col = getSky(rd, sunDir);
    } else {
        // 1. Intersect camera ray with the water surface (y = 0)
        float tWater = -ro.y / rd.y;
        vec3 pWater = ro + rd * tWater;
        
        // 2. Compute water surface normal
        float eps = 0.015 + tWater * 0.0012;
        float h0 = waterHeight(pWater.xz * 0.8);
        float hx = waterHeight((pWater.xz + vec2(eps, 0.0)) * 0.8);
        float hz = waterHeight((pWater.xz + vec2(0.0, eps)) * 0.8);
        vec3 n = normalize(vec3(h0 - hx, eps * 3.0, h0 - hz));
        
        // 3. Refraction: Bend the ray entering water (Air IOR 1.0 -> Water IOR 1.333)
        vec3 refrDir = refract(rd, n, 1.0 / 1.333);
        
        // Intersect refracted ray with an undulating seabed around y = -2.2
        float baseDepth = 2.2 + sin(pWater.x * 0.25) * 0.6 + cos(pWater.z * 0.2) * 0.6;
        float tFloor = baseDepth / max(-refrDir.y, 0.05);
        vec3 pFloor = pWater + refrDir * tFloor;
        
        // Render the refracted seabed + caustics
        vec3 refractedCol = renderSeabed(pFloor, sunDir, tFloor);
        
        // 4. Reflection: Sky & sun sparkles off the water surface
        vec3 reflDir = reflect(rd, n);
        reflDir.y = abs(reflDir.y);
        vec3 reflectedCol = getSky(reflDir, sunDir);
        
        // Crisp sun specular glints on wave crests
        vec3 halfVec = normalize(sunDir - rd);
        float spec = pow(max(dot(n, halfVec), 0.0), 400.0) * 3.5;
        reflectedCol += vec3(1.0, 0.96, 0.88) * spec;
        
        // 5. Schlick Fresnel: Look straight down to see the seabed, look grazing to see sky
        float fresnel = 0.02 + 0.98 * pow(1.0 - max(dot(-rd, n), 0.0), 5.0);
        col = mix(refractedCol, reflectedCol, clamp(fresnel, 0.0, 1.0));
        
        // Horizon haze blend
        float fog = 1.0 - exp(-tWater * 0.006);
        col = mix(col, getSky(vec3(rd.x, 0.001, rd.z), sunDir), fog);
    }
    
    // Tone mapping & gamma correction
    col = col / (1.0 + col * 0.25);
    col = pow(col, vec3(0.88));
    
    fragColor = vec4(col, 1.0);
}
