// --- PROCEDURAL HASH & NOISE ---
float hash(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
}

float hash3(vec3 p) {
    return fract(sin(dot(p, vec3(127.1, 311.7, 74.7))) * 43758.5453123);
}

float noise(vec2 p) {
    vec2 i = floor(p);
    vec2 f = fract(p);
    vec2 u = f * f * (3.0 - 2.0 * f);
    return mix(
        mix(hash(i + vec2(0.0, 0.0)), hash(i + vec2(1.0, 0.0)), u.x),
        mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), u.x),
        u.y
    );
}

float noise3(vec3 p) {
    vec3 i = floor(p);
    vec3 f = fract(p);
    vec3 u = f * f * (3.0 - 2.0 * f);
    return mix(
        mix(
            mix(hash3(i + vec3(0,0,0)), hash3(i + vec3(1,0,0)), u.x),
            mix(hash3(i + vec3(0,1,0)), hash3(i + vec3(1,1,0)), u.x),
            u.y
        ),
        mix(
            mix(hash3(i + vec3(0,0,1)), hash3(i + vec3(1,0,1)), u.x),
            mix(hash3(i + vec3(0,1,1)), hash3(i + vec3(1,1,1)), u.x),
            u.y
        ),
        u.z
    );
}

const mat2 m2 = mat2(0.8, -0.6, 0.6, 0.8);

// --- WATER SURFACE & CAUSTICS ---
float waterHeight(vec2 p) {
    float h = 0.0;
    float amp = 0.5;
    float freq = 1.0;
    for (int i = 0; i < 5; i++) {
        vec2 q = p * freq;
        float w1 = sin(q.x * 1.5 + u_time * 1.4 + noise(q + u_time * 0.4) * 2.0);
        float w2 = cos(q.y * 1.3 - u_time * 1.1 + noise(q - u_time * 0.3) * 2.0);
        h += amp * (1.0 - abs(w1 * w2));
        p = m2 * p;
        freq *= 1.85;
        amp *= 0.48;
    }
    return h;
}

float causticPattern(vec2 uv) {
    vec2 p = mod(uv * 6.28318, 6.28318) - 250.0;
    vec2 i = p;
    float c = 1.0;
    float inten = 0.005;
    for (int n = 0; n < 4; n++) {
        float t = u_time * (1.0 - (3.5 / float(n + 1)));
        i = p + vec2(cos(t - i.x) + sin(t + i.y), sin(t - i.y) + cos(t + i.x));
        c += 1.0 / length(vec2(p.x / (sin(i.x + t) / inten), p.y / (cos(i.y + t) / inten)));
    }
    c /= 4.0;
    c = 1.17 - pow(c, 1.4);
    return pow(abs(c), 8.0);
}

float getCaustics(vec3 pos, vec3 sunDir) {
    // Project 3D world position along the sun's rays onto the surface plane
    vec2 uv = (pos.xz - pos.y * (sunDir.xz / max(sunDir.y, 0.2))) * 0.22;
    float c1 = causticPattern(uv + vec2(u_time * 0.03, -u_time * 0.02));
    float c2 = causticPattern(uv * 1.35 - vec2(u_time * 0.025, u_time * 0.03));
    return clamp(c1 + c2 * 0.65, 0.0, 1.6);
}

// --- CORAL REEF & SEABED SDF ---
// Smooth minimum for organic coral/rock blending
float smin(float a, float b, float k) {
    float h = clamp(0.5 + 0.5 * (b - a) / k, 0.0, 1.0);
    return mix(b, a, h) - k * h * (1.0 - h);
}

// Returns vec2(signedDistance, materialID)
// materialID: 0.0 = sand, 1.0 = reef rock, 2.0 = vibrant coral
vec2 mapReef(vec3 p) {
    // Winding canyon channel so the camera swims between reef walls
    float canyon = sin(p.z * 0.14) * 2.8;
    float distFromCenter = abs(p.x - canyon);
    
    // Base sandy seabed at y = -4.5 with gentle dunes and ripples
    float sandHeight = -4.5 + sin(p.x * 0.4 + p.z * 0.2) * 0.35 
                     + sin(p.x * 3.5 + noise(p.xz * 2.0) * 2.0) * 0.04;
    // Raise terrain on the left and right edges to form reef ridges
    sandHeight += smoothstep(2.2, 8.5, distFromCenter) * 3.4;
    float dSand = p.y - sandHeight;
    
    // Large porous reef boulders along the ridge walls
    vec3 q = p;
    q.xz = mod(q.xz + 3.5, 7.0) - 3.5;
    float rockBaseY = sandHeight - 0.3;
    float dRock = length(vec3(q.x, (p.y - rockBaseY) * 1.15, q.z)) - 1.85;
    // Carve craggy rock crevices using 3D noise
    dRock += (noise3(p * 1.1) - 0.5) * 0.85 + (noise3(p * 3.2) - 0.5) * 0.25;
    // Keep the central swim path open
    dRock = max(dRock, 2.0 - distFromCenter);
    
    // Brain & staghorn coral clusters perched atop the rocks
    vec3 cq = p;
    cq.xz = mod(cq.xz + 1.75, 3.5) - 1.75;
    float coralY = rockBaseY + 1.35;
    float dCoral = length(cq - vec3(0.0, coralY, 0.0)) - 0.68;
    // Gyroid-like brain coral ridges & porous bumps
    float brainPattern = sin(p.x * 11.0 + sin(p.y * 9.0)) * cos(p.z * 11.0 + cos(p.x * 9.0));
    dCoral += brainPattern * 0.07 + (noise3(p * 5.0) - 0.5) * 0.14;
    dCoral = max(dCoral, 2.3 - distFromCenter);
    
    // Combine sand, rock, and coral smoothly
    float dTerrain = smin(dSand, dRock, 0.65);
    float matID = (dRock < dSand + 0.15) ? 1.0 : 0.0;
    
    if (dCoral < dTerrain) {
        matID = 2.0;
    }
    float dFinal = smin(dTerrain, dCoral, 0.18);
    
    return vec2(dFinal, matID);
}

vec3 calcNormal(vec3 p) {
    vec2 e = vec2(0.015, 0.0);
    return normalize(vec3(
        mapReef(p + e.xyy).x - mapReef(p - e.xyy).x,
        mapReef(p + e.yxy).x - mapReef(p - e.yxy).x,
        mapReef(p + e.yyx).x - mapReef(p - e.yyx).x
    ));
}

// --- SKY & SNELL'S WINDOW (LOOKING UP FROM UNDERWATER) ---
vec3 getSkyAbove(vec3 rd, vec3 sunDir) {
    float sunAmt = max(dot(rd, sunDir), 0.0);
    vec3 sky = mix(vec3(0.45, 0.78, 0.98), vec3(0.15, 0.45, 0.85), clamp(rd.y * 1.5, 0.0, 1.0));
    sky += vec3(1.0, 0.96, 0.85) * pow(sunAmt, 10.0) * 0.6;
    sky += vec3(1.0, 0.98, 0.92) * smoothstep(0.996, 0.999, sunAmt) * 4.0;
    return sky;
}

vec3 renderSurfaceFromBelow(vec3 ro, vec3 rd, vec3 sunDir) {
    // Intersect upward ray with ocean surface at y = 0
    float t = -ro.y / rd.y;
    vec3 p = ro + rd * t;
    
    float eps = 0.025 + t * 0.002;
    float h0 = waterHeight(p.xz * 0.65);
    float hx = waterHeight((p.xz + vec2(eps, 0.0)) * 0.65);
    float hz = waterHeight((p.xz + vec2(0.0, eps)) * 0.65);
    // Normal points downward into the water
    vec3 n = normalize(vec3(hx - h0, -eps * 2.2, hz - h0));
    
    // Snell's Law from water (IOR 1.333) to air (IOR 1.0): ratio = 1.333 / 1.0
    vec3 refrDir = refract(rd, n, 1.333);
    
    vec3 deepReflection = vec3(0.01, 0.14, 0.25);
    if (length(refrDir) < 0.001) {
        // Total Internal Reflection (outside Snell's Window)
        return deepReflection;
    }
    
    // Inside Snell's Window: bright view of the sky and sun refracted through ripples
    float cosTheta = max(dot(-rd, n), 0.0);
    float fresnel = pow(1.0 - cosTheta, 3.0);
    vec3 skyCol = getSkyAbove(refrDir, sunDir);
    return mix(skyCol, deepReflection, clamp(fresnel, 0.0, 1.0));
}

// --- VOLUMETRIC GOD RAYS & SUSPENDED PLANKTON ---
vec3 renderGodRays(vec3 ro, vec3 rd, float maxT, vec3 sunDir) {
    int steps = 24;
    float dt = min(maxT, 32.0) / float(steps);
    float t = dt * 0.5;
    vec3 rays = vec3(0.0);
    
    for (int i = 0; i < 24; i++) {
        vec3 p = ro + rd * t;
        // Sample caustics near the surface along the sun direction to create shafts of light
        float beam = getCaustics(vec3(p.x, 0.0, p.z), sunDir);
        // Beams fade naturally with water depth
        float depthAtten = exp(p.y * 0.22);
        rays += vec3(0.18, 0.52, 0.62) * beam * depthAtten * dt * 0.045;
        t += dt;
    }
    return rays;
}

// --- MAIN RENDER ---
void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    vec2 uv = (fragCoord * 2.0 - u_resolution.xy) / u_resolution.y;
    
    // Camera swimming gently forward through the coral canyon at depth y = -2.2
    float zPos = u_time * 1.8;
    float canyonTrack = sin(zPos * 0.14) * 2.2;
    vec3 ro = vec3(canyonTrack + sin(u_time * 0.6) * 0.25, -2.1 + cos(u_time * 0.5) * 0.15, zPos);
    
    // Slight upward tilt so both the reef floor and shimmering surface are visible
    vec3 rd = normalize(vec3(uv.x, uv.y + 0.08, 1.45));
    
    vec3 sunDir = normalize(vec3(0.25, 0.82, 0.52));
    vec3 waterFogColor = vec3(0.015, 0.22, 0.36);
    
    // Raymarch the coral reef and seabed
    float t = 0.0;
    float maxDist = 48.0;
    vec2 hit = vec2(0.0);
    bool hitReef = false;
    
    for (int i = 0; i < 85; i++) {
        vec3 p = ro + rd * t;
        if (p.y > 0.05 && rd.y > 0.0) break; // Ray exited water surface
        
        hit = mapReef(p);
        if (hit.x < 0.012) {
            hitReef = true;
            break;
        }
        t += hit.x * 0.75;
        if (t > maxDist) break;
    }
    
    vec3 col = waterFogColor;
    float effectiveDist = maxDist;
    
    if (hitReef) {
        effectiveDist = t;
        vec3 p = ro + rd * t;
        vec3 n = calcNormal(p);
        
        // Material coloring
        vec3 albedo = vec3(0.84, 0.78, 0.62); // Warm coral sand
        if (hit.y > 1.5) {
            // Vibrant coral palette (magenta, tangerine, seafoam) based on position
            float hueSeed = noise(floor(p.xz * 0.5));
            vec3 coralA = vec3(0.92, 0.26, 0.45); // Pink/magenta
            vec3 coralB = vec3(0.95, 0.52, 0.18); // Tangerine orange
            vec3 coralC = vec3(0.68, 0.35, 0.85); // Violet
            albedo = mix(coralA, coralB, smoothstep(0.2, 0.6, hueSeed));
            albedo = mix(albedo, coralC, smoothstep(0.6, 0.9, hueSeed));
            albedo *= 0.85 + 0.25 * noise3(p * 8.0);
        } else if (hit.y > 0.5) {
            // Weathered limestone reef rock with algae patches
            float moss = smoothstep(0.45, 0.75, noise3(p * 1.8));
            albedo = mix(vec3(0.32, 0.34, 0.36), vec3(0.22, 0.42, 0.28), moss);
            albedo *= 0.75 + 0.4 * noise3(p * 6.0);
        }
        
        // Lighting: diffuse sunlight + ambient bounce + animated caustics
        float diff = max(dot(n, sunDir), 0.0);
        float ambient = 0.25 + 0.25 * n.y;
        float caustics = getCaustics(p, sunDir) * smoothstep(-0.2, 0.6, n.y);
        
        vec3 litSurface = albedo * (diff * 0.85 + ambient) + vec3(0.75, 0.98, 1.0) * caustics * 0.95;
        
        // Beer-Lambert underwater depth & distance color absorption
        vec3 absorption = exp(-vec3(0.14, 0.045, 0.025) * t);
        col = mix(waterFogColor, litSurface * absorption, exp(-t * 0.042));
        
    } else if (rd.y > 0.0) {
        // Looking up through the ocean surface (Snell's Window)
        float surfDist = -ro.y / rd.y;
		effectiveDist = min(surfDist, maxDist);
		vec3 surfCol = renderSurfaceFromBelow(ro, rd, sunDir);
		col = mix(waterFogColor, surfCol, exp(-effectiveDist * 0.045));
	}
	
	// Add volumetric sunbeams (god rays) piercing down from the surface
	col += renderGodRays(ro, rd, effectiveDist, sunDir);
	
	// Subtle floating marine snow / plankton particles
	vec3 pGrid = fract((ro + rd * 4.0) * 1.8 + vec3(0.0, -u_time * 0.15, 0.0)) - 0.5;
	float specks = smoothstep(0.06, 0.01, length(pGrid)) * 0.25;
	col += vec3(0.5, 0.85, 0.95) * specks;
	
	// Film tone mapping & gamma correction
	col = col / (1.0 + col * 0.25);
	col = pow(col, vec3(0.9));
	fragColor = vec4(col, 1.0);
}
