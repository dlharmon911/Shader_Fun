#define NSHAPES 200.0
#define NPOINTS 100.0 // Reduced for performance; GLSL loops can be heavy
#define PI 3.14159265359

// Simple HSB to RGB conversion
vec3 hsb2rgb(vec3 c)
{
    vec4 K = vec4(1.0, 2.0 / 3.0, 1.0 / 3.0, 3.0);
    vec3 p = abs(fract(c.xxx + K.xyz) * 6.0 - K.www);
    return c.z * mix(K.xxx, clamp(p - K.xxx, 0.0, 1.0), c.y);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    // Map screen coordinates to -2.0 to 2.0 range (matching your MAXY)
    vec2 uv = (fragCoord.xy - 0.5 * u_resolution) / u_resolution.y * 4.0;
    float t = u_time * 0.2; // Slowed down DT
    float R = 2.0 * PI / NSHAPES;
    vec3 finalCol = vec3(0.0);
    float x = 0.0;
    float y = 0.0;

    for (float i = 0.0; i < NSHAPES; i++) {
        float shape_t = R * i + t;
        
        for (float j = 0.0; j < NPOINTS; j++) {
            float ang_y = i + y;
            float ang_x = shape_t + x;
            
            x = sin(ang_y) + sin(ang_x);
            y = cos(ang_y) + cos(ang_x);
            
            // Check distance from pixel to current point (x, y)
            float dist = distance(uv, vec2(x, y));
            
            // Draw a soft "glow" point
            if (dist < 0.03) {
                vec3 pntCol = hsb2rgb(vec3(i / NSHAPES, 0.8, 0.9));
                finalCol += pntCol * (0.002 / dist); // Additive blending
            }
        }
    }
    fragColor = vec4(finalCol, 1.0);
}
