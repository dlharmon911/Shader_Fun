// --- PROCEDURAL NOISE & FBM ---
float hash(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
}

float noise(vec2 p) {
    vec2 i = floor(p);
    vec2 f = fract(p);
    vec2 u = f * f * (3.0 - 2.0 * f);
    return mix(
        mix(hash(i + vec2(0.0, 0.0)), hash(i + vec2(1.0, 0.0)), u.x),
        mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), u.x),
        u.y
    );
}

const mat2 m2 = mat2(0.8, -0.6, 0.6, 0.8);

// Multi-octave water wave heightfield
float waterHeight(vec2 p) {
    float h = 0.0;
    float amp = 0.5;
    float freq = 1.0;
    
    for (int i = 0; i < 6; i++) {
        vec2 q = p * freq;
        float w1 = sin(q.x * 1.5 + u_time * 1.4 + noise(q + u_time * 0.4) * 2.0);
        float w2 = cos(q.y * 1.3 - u_time * 1.1 + noise(q - u_time * 0.3) * 2.0);
        h += amp * (1.0 - abs(w1 * w2));
        
        p = m2 * p;
        freq *= 1.85;
        amp *= 0.48;
    }
    return h;
}

// Procedural Voronoi-like cellular caustics pattern
float causticPattern(vec2 uv) {
    vec2 p = mod(uv * 6.28318, 6.28318) - 250.0;
    vec2 i = p;
    float c = 1.0;
    float inten = 0.005;

    for (int n = 0; n < 4; n++) {
        float t = u_time * (1.0 - (3.5 / float(n + 1)));
        i = p + vec2(
            cos(t - i.x) + sin(t + i.y),
            sin(t - i.y) + cos(t + i.x)
        );
        c += 1.0 / length(vec2(
            p.x / (sin(i.x + t) / inten),
            p.y / (cos(i.y + t) / inten)
        ));
    }
    c /= 4.0;
    c = 1.17 - pow(c, 1.4);
    return pow(abs(c), 8.0);
}

// Multi-layered animated caustics projected onto the seabed
float getCaustics(vec2 pos, vec3 sunDir) {
    // Offset coordinates by sun angle so caustics align with light rays
    vec2 uv = pos * 0.25 - sunDir.xz * 0.5;
    float c1 = causticPattern(uv + vec2(u_time * 0.03, -u_time * 0.02));
    float c2 = causticPattern(uv * 1.35 - vec2(u_time * 0.025, u_time * 0.03));
    return clamp(c1 + c2 * 0.6, 0.0, 1.5);
}

// Procedural sandy seabed height & relief
float seabedHeight(vec2 p) {
    float dunes = sin(p.x * 0.35 + noise(p * 0.2) * 3.0) * 0.6;
    float ripples = sin(p.x * 3.0 + p.y * 1.5 + noise(p * 1.5) * 2.0) * 0.06;
    float detail = (noise(p * 1.2) - 0.5) * 0.8;
    // Base depth around y = -4.2
    return -4.2 + dunes + ripples + detail;
}

// --- SKY & CLOUDS ---
float cloudDensity(vec3 p) {
    vec2 uv = p.xz * 0.1 + vec2(u_time * 0.025, u_time * 0.012);
    float f = 0.0;
    float amp = 0.5;
    for (int i = 0; i < 4; i++) {
        f += amp * noise(uv);
        uv = m2 * uv * 2.05;
        amp *= 0.5;
    }
    float heightMask = smoothstep(18.0, 23.0, p.y) * smoothstep(38.0, 27.0, p.y);
    return max(0.0, (f - 0.46) * heightMask * 2.5);
}

vec3 getSky(vec3 rd, vec3 sunDir) {
    float sunAmt = max(dot(rd, sunDir), 0.0);
    // Bright tropical midday sky
    vec3 sky = mix(vec3(0.42, 0.72, 0.96), vec3(0.12, 0.38, 0.78), clamp(rd.y * 1.6, 0.0, 1.0));
    sky = mix(sky, vec3(0.75, 0.90, 0.98), pow(1.0 - max(rd.y, 0.0), 5.0));
    
    sky += vec3(1.0, 0.95, 0.85) * pow(sunAmt, 12.0) * 0.45;
    sky += vec3(1.0, 0.98, 0.90) * smoothstep(0.998, 0.9995, sunAmt) * 3.5;
    return sky;
}

vec4 renderClouds(vec3 ro, vec3 rd, vec3 sunDir) {
    if (rd.y <= 0.01) return vec4(0.0);
    float tMin = max(0.0, (18.0 - ro.y) / rd.y);
    float tMax = min(220.0, (38.0 - ro.y) / rd.y);
    if (tMin >= tMax) return vec4(0.0);
    
    float dt = (tMax - tMin) / 24.0;
    float t = tMin;
    vec3 sumCol = vec3(0.0);
    float transmittance = 1.0;
    
    for (int i = 0; i < 24; i++) {
        if (transmittance < 0.02) break;
        vec3 p = ro + rd * t;
        float dens = cloudDensity(p);
        if (dens > 0.001) {
            float shadowDens = cloudDensity(p + sunDir * 2.5);
            float light = exp(-shadowDens * 1.5);
            vec3 cloudCol = mix(vec3(0.65, 0.75, 0.88), vec3(1.0, 0.99, 0.96), light);
            float alpha = 1.0 - exp(-dens * dt * 0.45);
            sumCol += transmittance * alpha * cloudCol;
            transmittance *= (1.0 - alpha);
        }
        t += dt;
    }
    float horizonFade = smoothstep(0.01, 0.16, rd.y);
    return vec4(sumCol * horizonFade, (1.0 - transmittance) * horizonFade);
}

// --- UNDERWATER REFRACTION & CAUSTICS ---
vec3 renderUnderwater(vec3 surfacePos, vec3 refrDir, vec3 sunDir) {
    // Raymarch beneath the surface until hitting the sandy seabed
    float t = 0.0;
    float maxDist = 35.0;
    vec3 p = surfacePos;
    bool hit = false;
    
    for (int i = 0; i < 36; i++) {
        p = surfacePos + refrDir * t;
        float bedY = seabedHeight(p.xz);
        float d = p.y - bedY;
        if (d < 0.04) {
            hit = true;
            break;
        }
        t += max(d * 0.65, 0.12);
        if (t > maxDist) break;
    }
    
    // Deep ocean scatter color if ray never hits shallow sand
    vec3 deepColor = vec3(0.01, 0.16, 0.28);
    if (!hit) return deepColor;
    
    // Seabed normal for sand dune shading
    float eps = 0.08;
    float h0 = seabedHeight(p.xz);
    float hx = seabedHeight(p.xz + vec2(eps, 0.0));
    float hz = seabedHeight(p.xz + vec2(0.0, eps));
    vec3 bedNorm = normalize(vec3(h0 - hx, eps, h0 - hz));
    
    // Refracted sun direction entering water (IOR 1.0 -> 1.333)
    vec3 refrSun = refract(-sunDir, vec3(0.0, 1.0, 0.0), 0.75);
    float diff = max(dot(bedNorm, -refrSun), 0.15);
    
    // Warm golden sand color with subtle dark mineral flecks
    float grain = noise(p.xz * 18.0) * 0.12;
    vec3 sandBase = vec3(0.86, 0.79, 0.62) - grain;
    
    // Add bright dancing caustics focused onto the seabed
    float caustics = getCaustics(p.xz, sunDir);
    vec3 causticLight = vec3(0.95, 1.0, 0.92) * caustics * 1.45;
    
    vec3 litSand = sandBase * diff + causticLight;
    
    // Beer-Lambert Law: red light absorbs fastest in water, leaving turquoise-blue
    float depth = abs(p.y);
    float pathLen = t;
    vec3 absorption = exp(-vec3(0.38, 0.11, 0.06) * (pathLen + depth * 0.5));
    
    return mix(deepColor, litSand * absorption, exp(-pathLen * 0.045));
}

// --- MAIN RENDER ---
void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    vec2 uv = (fragCoord * 2.0 - u_resolution.xy) / u_resolution.y;
    
    vec3 ro = vec3(0.0, 3.6, u_time * 2.2);
    vec3 rd = normalize(vec3(uv.x, uv.y - 0.14, 1.5));
    
    // High afternoon sun for bright underwater caustics
    vec3 sunDir = normalize(vec3(0.3, 0.68, 0.7));
    
    vec3 col = vec3(0.0);
    
    if (rd.y >= 0.0) {
        col = getSky(rd, sunDir);
        vec4 clouds = renderClouds(ro, rd, sunDir);
        col = mix(col, clouds.rgb, clouds.a);
    } else {
        // Intersect water plane at y = 0
        float t = -ro.y / rd.y;
        vec3 p = ro + rd * t;
        
        // Wave surface normal
        float eps = 0.02 + t * 0.0015;
        float h0 = waterHeight(p.xz * 0.65);
        float hx = waterHeight((p.xz + vec2(eps, 0.0)) * 0.65);
        float hz = waterHeight((p.xz + vec2(0.0, eps)) * 0.65);
        vec3 n = normalize(vec3(h0 - hx, eps * 2.4, h0 - hz));
        
        // Schlick Fresnel (water IOR = 1.333 -> F0 = 0.02)
        float cosTheta = max(dot(-rd, n), 0.0);
        float fresnel = 0.02 + 0.98 * pow(1.0 - cosTheta, 5.0);
        
        // 1. Reflection (Sky & Clouds)
        vec3 reflDir = reflect(rd, n);
        reflDir.y = abs(reflDir.y);
        vec3 reflCol = getSky(reflDir, sunDir);
        vec4 reflClouds = renderClouds(p, reflDir, sunDir);
        reflCol = mix(reflCol, reflClouds.rgb, reflClouds.a * 0.75);
        
        // 2. Refraction (Snell's Law bending into the seabed)
        // Only compute detailed seabed refraction up to 90 units out for crisp performance
        vec3 refrCol = vec3(0.01, 0.16, 0.28);
        if (t < 90.0) {
            vec3 refrDir = refract(rd, n, 1.0 / 1.333);
            refrCol = renderUnderwater(p, refrDir, sunDir);
        }
        
        // Combine refracted seabed and reflected sky via Fresnel
        col = mix(refrCol, reflCol, fresnel);
        
        // Bright solar specular glints on wave crests
        vec3 halfVec = normalize(sunDir - rd);
        float spec = pow(max(dot(n, halfVec), 0.0), 380.0) * 5.0;
        col += vec3(1.0, 0.98, 0.92) * spec;
        
        // Horizon atmospheric haze
        float fog = 1.0 - exp(-t * 0.004);
        col = mix(col, getSky(vec3(rd.x, 0.001, rd.z), sunDir), fog);
    }
    
    // Tone mapping & gamma correction
    col = col / (1.0 + col * 0.28);
    col = pow(col, vec3(0.88));
    
    fragColor = vec4(col, 1.0);
}
